cmake_minimum_required(VERSION 3.16)
project(MiniVectorDB VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(minivectordb_core src/db.cpp src/disk_store.cpp src/ivf_index.cpp)
target_include_directories(minivectordb_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(minivectordb_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(minivectordb src/main.cpp src/legacy_main.cpp)
add_executable(minivectordb_legacy src/legacy_main.cpp)
target_compile_definitions(minivectordb_legacy PRIVATE MVDB_LEGACY_DEFAULT)
target_link_libraries(minivectordb_legacy PRIVATE minivectordb_core)
target_link_libraries(minivectordb PRIVATE minivectordb_core)
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME json_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_json_protocol.py $<TARGET_FILE:minivectordb>)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME fault_injection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fault_injection.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME randomized_differential COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_randomized.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME disk_btree_stress COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_disk_storage.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME bounded_memory COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bounded_memory.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME wal_rotation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_wal_rotation.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME ann_transactional COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ann_index.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME compaction_and_crash COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_compaction.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME vector_scan_cache COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_vector_scan_cache.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME deep_btree COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_deep_btree.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME pk_btree_range COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pk_range.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME mapped_vector_forced COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mapped_scan_forced.py $<TARGET_FILE:minivectordb_legacy>)
  add_test(NAME mapped_vector_large COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_large_scan.py $<TARGET_FILE:minivectordb_legacy> --rows 25000 --queries 8 --varchar --output ${CMAKE_CURRENT_BINARY_DIR}/test_mapped_scan.json)
endif()
